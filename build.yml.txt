name: Build FreeGram
on:
  push:
  workflow_dispatch:
jobs:
  android:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: {node-version: 20}
      - uses: actions/setup-java@v4
        with: {distribution: temurin, java-version: 17}
      - name: Build APK
        run: |
          sed -i "s/__BUILD__/${{ github.run_number }}/" index.html
          mkdir -p www
          cp index.html icon.png www/
          npm install
          npx cap add android
          pip install pillow
          python3 make-icons.py android
          npx cap sync android
          cd android
          chmod +x gradlew
          ./gradlew assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: FreeGram-APK
          path: android/app/build/outputs/apk/debug/app-debug.apk
  windows:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: {node-version: 20}
      - name: Build EXE
        shell: bash
        env:
          CSC_IDENTITY_AUTO_DISCOVERY: "false"
        run: |
          sed -i "s/__BUILD__/${{ github.run_number }}/" index.html
          mkdir -p www
          cp index.html icon.png www/
          npm install
          npm run dist
      - uses: actions/upload-artifact@v4
        with:
          name: FreeGram-EXE
          path: dist/FreeGram.exe
  release:
    needs: [android, windows]
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - uses: actions/download-artifact@v4
        with:
          pattern: FreeGram-*
          merge-multiple: true
          path: out
      - run: mv out/app-debug.apk out/FreeGram.apk
      - uses: softprops/action-gh-release@v2
        with:
          tag_name: v${{ github.run_number }}
          name: FreeGram ${{ github.run_number }}
          files: |
            out/FreeGram.apk
            out/FreeGram.exe
